`Detector.swift`：

```swift
public struct DetectContext {
    public let files: FileOperations
    public let safety: PathSafety
    public let runner: ProcessRunner
    public let now: Date
    public let cancellation: OperationCancellation?

    public init(files: FileOperations, safety: PathSafety, runner: ProcessRunner, now: Date,
                cancellation: OperationCancellation? = nil) {
        self.files = files
        self.safety = safety
        self.runner = runner
        self.now = now
        self.cancellation = cancellation
    }

    public var home: String { files.homeDirectory() }
    public var isCancelled: Bool { cancellation?.isCancelled == true }
}

/// 识别器只读：只能查看文件与运行只读命令，产出清理项，不做任何修改。
public protocol Detector {
    /// 扫描进度里显示的名称
    var name: String { get }
    func detect(_ context: DetectContext) -> [CleanItem]
}

public enum ScanStep: Equatable, Sendable {
    case detecting(String)
    case sizing(done: Int, total: Int)
    case spaceMap
    case checkingBlockers

    public var title: String {
        switch self {
        case .detecting(let name): return "正在识别\(name)"
        case .sizing(let done, let total): return "正在计算大小（\(done)/\(total)）"
        case .spaceMap: return "正在查看其他大目录"
        case .checkingBlockers: return "正在检查占用"
        }
    }
}
```

各识别器加 `public var name: String`：`RuleDetector` 为「规则」，`ProjectArtifactDetector` 为「项目」，`DockerDetector` 为「Docker」。

`Sizer`：增加 `public let cancellation: OperationCancellation?`，`init(files: FileOperations, cancellation: OperationCancellation? = nil)`；`measure` 的遍历循环每处理一个路径先检查 `cancellation?.isCancelled == true`，是则结束遍历并返回已累计的结果。

    /// 返回超时仍未退出的应用，由界面逐个询问是否强制退出。编辑器与 IDE（canQuitForUser == false）一律不处理。
    public func quitPhase(_ items: [CleanItem], consentedApps: Set<String>, timeout: TimeInterval = 10) -> [QuitApp] {
        var seen = Set<String>()
        var stuck: [QuitApp] = []
        for app in items.filter(willRun).flatMap({ $0.preconditions.quitApps ?? [] }) where seen.insert(app.bundleID).inserted {
            guard apps.isRunning(bundleID: app.bundleID), app.canQuitForUser, consentedApps.contains(app.bundleID) else { continue }
            guard (try? audit.appendRequired(AuditEvent(op: "run.app.quit", detail: app.bundleID))) != nil else { continue }
            if !apps.requestQuit(bundleID: app.bundleID, timeout: timeout) { stuck.append(app) }
        }
        return stuck
    }

    /// 用户对这些应用逐个同意了强制退出。
    public func forceQuit(_ targets: [QuitApp]) {
        for app in targets where app.canQuitForUser && apps.isRunning(bundleID: app.bundleID) {
            guard (try? audit.appendRequired(AuditEvent(op: "run.app.force-quit", detail: app.bundleID))) != nil else { continue }
            _ = apps.forceQuit(bundleID: app.bundleID)
        }
    }

import Foundation
import CleanerCore

public struct RecoveryOverview: Equatable {
    public struct Item: Equatable, Identifiable {
        public var id: String
        public var title: String
        public var bytes: Int64
        /// 条目记录了所属清理项才能单独还原；否则只能整批还原。
        public var restorableIndividually: Bool
    }

    public struct Batch: Equatable, Identifiable {
        public var id: String
        public var createdAt: Date
        public var bytes: Int64
        public var daysLeft: Int
        public var items: [Item]
    }

    public var batches: [Batch]
    public var orphans: [String]

    public static let empty = RecoveryOverview(batches: [], orphans: [])

    public var totalBytes: Int64 { batches.reduce(0) { $0 + $1.bytes } }
    public var expiredBatchIDs: [String] { batches.filter { $0.daysLeft <= 0 }.map(\.id) }

    public static func make(batches: [RecoveryBatch], orphans: [String], occupied: (RecoveryBatch) -> Int64,
                            now: Date, retentionDays: Int = RecoveryStore.retentionDays) -> RecoveryOverview {
        let rows = batches.sorted { $0.createdAt > $1.createdAt }.compactMap { batch -> Batch? in
            var order: [String] = []
            var titles: [String: String] = [:]
            var bytes: [String: Int64] = [:]
            var individual: [String: Bool] = [:]
            for entry in batch.entries where entry.state == .stored || entry.state == .pending {
                let key = entry.itemID ?? entry.originalPath
                if titles[key] == nil {
                    order.append(key)
                    titles[key] = entry.itemTitle ?? (entry.originalPath as NSString).lastPathComponent
                    individual[key] = entry.itemID != nil
                }
                bytes[key, default: 0] += entry.bytes
            }
            guard !order.isEmpty else { return nil }
            let age = Int(now.timeIntervalSince(batch.createdAt) / 86_400)
            return Batch(id: batch.id, createdAt: batch.createdAt, bytes: occupied(batch),
                         daysLeft: max(retentionDays - age, 0),
                         items: order.map { Item(id: $0, title: titles[$0]!, bytes: bytes[$0]!, restorableIndividually: individual[$0]!) })
        }
        return RecoveryOverview(batches: rows, orphans: orphans)
    }
}

import Foundation
import CleanerCore

public struct ResultSummary: Equatable {
    public struct Group: Equatable {
        public var reason: OutcomeReason
        public var results: [ItemRunResult]
    }

    public var report: RunReport
    public var doneCount: Int
    public var skippedGroups: [Group]
    public var failedGroups: [Group]
    /// 实际腾出远小于「工具自带命令」项的预估时给出的解释；移入恢复区的项本来就不腾出空间，不计入预估。
    public var gapHint: String?

    public init(report: RunReport, commandItemIDs: Set<String>) {
        self.report = report
        doneCount = report.results.filter { $0.outcome == .done }.count
        skippedGroups = Self.group(report.results) { if case .skipped = $0 { return true }; return false }
        failedGroups = Self.group(report.results) { if case .failed = $0 { return true }; return false }
        let expected = report.results.filter { $0.outcome == .done && commandItemIDs.contains($0.itemID) }
            .reduce(0) { $0 + $1.estimatedBytes }
        let freed = max(report.measuredFreedBytes ?? 0, 0)
        if expected > 100 << 20 && freed < expected / 2 {
            gapHint = "实际腾出比预估少很多，可能是：清理命令只删掉了一部分缓存、缓存文件与其他目录共用（硬链接），或清理期间其他程序在写入磁盘。"
        } else {
            gapHint = nil
        }
    }

    private static func group(_ results: [ItemRunResult], _ matches: (RunOutcome) -> Bool) -> [Group] {
        var order: [OutcomeReason] = []
        var buckets: [OutcomeReason: [ItemRunResult]] = [:]
        for result in results where matches(result.outcome) {
            let reason = result.reason ?? .other
            if buckets[reason] == nil { order.append(reason) }
            buckets[reason, default: []].append(result)
        }
        return order.map { Group(reason: $0, results: buckets[$0]!) }
    }
}

// Sources/AppCore/AppModel.swift
import Foundation
import Combine
import CleanerCore

/// 界面唯一的状态。约定：所有 @Published 属性只在主线程修改（经 main 回调）；引擎调用经 background 在后台执行。
public final class AppModel: ObservableObject {
    public enum Phase: Equatable {
        case idle, scanning, ready, confirming, executing, finished
    }

    public struct ExecutionProgress: Equatable {
        public var total = 0
        public var finished: [ItemRunResult] = []
        public var currentTitle: String?
        public var stopping = false
    }

    /// 扫描结果超过这个时间就先重新扫描再弹确认单
    public static let staleAfter: TimeInterval = 600

    @Published public private(set) var phase: Phase = .idle
    @Published public private(set) var scanStep: ScanStep?
    @Published public private(set) var scanResult: ScanResult?
    @Published public private(set) var selectedIDs: Set<String> = []
    @Published public private(set) var confirmation: ConfirmationSummary?
    @Published public private(set) var progress = ExecutionProgress()
    @Published public private(set) var stuckApps: [QuitApp] = []
    @Published public private(set) var report: RunReport?
    @Published public private(set) var recovery = RecoveryOverview.empty
    @Published public private(set) var message: String?

    private let engine: EngineServicing
    private let now: () -> Date
    private let background: (@escaping () -> Void) -> Void
    private let main: (@escaping () -> Void) -> Void
    private var scanCancellation: OperationCancellation?
    private var runCancellation: OperationCancellation?
    private var confirmAfterScan = false
    private var runningItems: [CleanItem] = []

    public init(engine: EngineServicing,
                now: @escaping () -> Date = { Date() },
                background: @escaping (@escaping () -> Void) -> Void = { DispatchQueue.global(qos: .userInitiated).async(execute: $0) },
                main: @escaping (@escaping () -> Void) -> Void = { DispatchQueue.main.async(execute: $0) }) {
        self.engine = engine
        self.now = now
        self.background = background
        self.main = main
    }

    public var plan: CleanPlan? { scanResult.map { CleanPlan(items: $0.items, selectedIDs: selectedIDs) } }
    public var defaultPlan: CleanPlan? { scanResult.map(PlanBuilder.build(from:)) }
    public var isStale: Bool {
        guard let scanResult else { return true }
        return now().timeIntervalSince(scanResult.generatedAt) > Self.staleAfter
    }
    public var resultSummary: ResultSummary? {
        report.map { report in
            ResultSummary(report: report, commandItemIDs: Set(runningItems.filter { $0.action.type == .command }.map(\.id)))
        }
    }

    public func clearMessage() { message = nil }

    // MARK: 扫描

    public func startScan() {
        guard phase != .scanning, phase != .executing, phase != .confirming else { return }
        let cancellation = OperationCancellation()
        scanCancellation = cancellation
        phase = .scanning
        scanStep = nil
        background { [engine] in
            do {
                let result = try engine.scan(onStep: { step in self.main { self.scanStep = step } }, cancellation: cancellation)
                self.main { self.finishScan(result) }
            } catch {
                self.main { self.failScan(error) }
            }
        }
    }

    public func cancelScan() { scanCancellation?.cancel() }

    private func finishScan(_ result: ScanResult) {
        scanCancellation = nil
        scanStep = nil
        if result.cancelled {
            confirmAfterScan = false
            phase = scanResult == nil ? .idle : .ready
            return
        }
        scanResult = result
        selectedIDs = PlanBuilder.build(from: result).selectedIDs
        phase = .ready
        if confirmAfterScan {
            confirmAfterScan = false
            presentConfirmation()
        }
    }

    private func failScan(_ error: Error) {
        scanCancellation = nil
        scanStep = nil
        confirmAfterScan = false
        message = String(describing: error)
        phase = scanResult == nil ? .idle : .ready
    }

    // MARK: 方案与确认

    public func toggle(_ id: String) {
        guard phase == .ready, let item = scanResult?.items.first(where: { $0.id == id }), item.tier != .displayOnly else { return }
        if selectedIDs.contains(id) { selectedIDs.remove(id) } else { selectedIDs.insert(id) }
    }

    public func requestClean(useDefaultSelection: Bool) {
        guard phase == .ready || phase == .idle || phase == .finished else { return }
        if isStale {
            confirmAfterScan = true
            if phase == .finished { phase = .ready }
            startScan()
            return
        }
        if useDefaultSelection, let defaultPlan { selectedIDs = defaultPlan.selectedIDs }
        presentConfirmation()
    }

    private func presentConfirmation() {
        guard let plan, !plan.selectedItems.filter({ $0.tier != .displayOnly }).isEmpty else {
            message = "没有选中可执行的清理项"
            return
        }
        confirmation = ConfirmationSummary.make(items: plan.selectedItems, isRunning: engine.isAppRunning)
        phase = .confirming
    }

    public func cancelConfirmation() {
        guard phase == .confirming else { return }
        confirmation = nil
        phase = .ready
    }

    // MARK: 执行

    public func startClean(consentedApps: Set<String>) {
        guard phase == .confirming, let confirmation else { return }
        let items = confirmation.items
        runningItems = items
        self.confirmation = nil
        report = nil
        progress = ExecutionProgress(total: items.count)
        let cancellation = OperationCancellation()
        runCancellation = cancellation
        phase = .executing
        background { [engine] in
            let stuck = engine.quitPhase(items, consentedApps: consentedApps)
            if stuck.isEmpty {
                self.runItems(items, cancellation: cancellation)
            } else {
                self.main { self.stuckApps = stuck }
            }
        }
    }

    /// 用户对超时未退出的应用逐个做了选择：勾选的强制退出，其余跳过相关项。
    public func resolveStuckApps(forceQuit: Set<String>) {
        guard phase == .executing, let cancellation = runCancellation else { return }
        let targets = stuckApps.filter { forceQuit.contains($0.bundleID) }
        stuckApps = []
        let items = runningItems
        background { [engine] in
            if !targets.isEmpty { engine.forceQuit(targets) }
            self.runItems(items, cancellation: cancellation)
        }
    }

    /// 在后台线程调用
    private func runItems(_ items: [CleanItem], cancellation: OperationCancellation) {
        do {
            let report = try engine.run(items, cancellation: cancellation,
                                        onItemStart: { index in self.main { self.progress.currentTitle = items[index].title } },
                                        onItemFinish: { _, result in self.main { self.progress.finished.append(result) } })
            main {
                self.report = report
                self.progress.currentTitle = nil
                self.phase = .finished
                self.refreshRecovery()
            }
        } catch {
            main {
                self.message = String(describing: error)
                self.progress.currentTitle = nil
                self.phase = .ready
            }
        }
    }

    public func stopClean() {
        guard phase == .executing else { return }
        runCancellation?.cancel()
        progress.stopping = true
    }

    /// 结果页点「完成」：回到方案页并重新扫描
    public func finish() {
        guard phase == .finished else { return }
        report = nil
        runCancellation = nil
        phase = .ready
        startScan()
    }

    // MARK: 恢复区

    public func refreshRecovery() {
        background { [engine, now] in
            let overview = RecoveryOverview.make(batches: engine.recoveryBatches(), orphans: engine.recoveryOrphans(),
                                                 occupied: engine.recoveryOccupied, now: now())
            self.main { self.recovery = overview }
        }
    }

    public func restore(batchID: String, itemIDs: Set<String>?) {
        guard phase != .executing else { return }
        background { [engine] in
            let result = engine.restore(batchID: batchID, itemIDs: itemIDs)
            self.main {
                self.message = result.errors.isEmpty
                    ? "已还原 \(result.restored.count) 个位置"
                    : "部分未能还原：\n" + result.errors.joined(separator: "\n")
                self.refreshRecovery()
            }
        }
    }

    public func purge(batchID: String) {
        guard phase != .executing else { return }
        background { [engine] in
            let errors = engine.purge(batchID: batchID)
            self.main {
                if !errors.isEmpty { self.message = errors.joined(separator: "\n") }
                self.refreshRecovery()
            }
        }
    }

    public func purgeExpired() {
        guard phase != .executing else { return }
        background { [engine, now] in
            let result = engine.purgeExpired(now: now())
            self.main {
                if !result.errors.isEmpty { self.message = result.errors.joined(separator: "\n") }
                self.refreshRecovery()
            }
        }
    }
}
